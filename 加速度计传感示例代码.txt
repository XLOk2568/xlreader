Wear OS 阅读器体感翻页方案（低功耗版）
作者：Copilot
方案：仅使用加速度计获取倾斜角度
优点：
- 功耗最低
- 不需要陀螺仪
- 不需要 Rotation Vector
- 适合阅读器
- 支持翻页/菜单/退出

========================
SensorTiltController.kt
========================

package com.xxx.reader.sensor

import android.content.Context
import android.hardware.Sensor
import android.hardware.SensorEvent
import android.hardware.SensorEventListener
import android.hardware.SensorManager
import kotlin.math.abs
import kotlin.math.atan2

class SensorTiltController(
    context: Context,
    private val onNextPage: () -> Unit,
    private val onPrevPage: () -> Unit,
    private val onOpenMenu: () -> Unit,
    private val onExitReader: () -> Unit,
) : SensorEventListener {

    private val sensorManager =
        context.getSystemService(Context.SENSOR_SERVICE)
                as SensorManager

    private val accelerometer =
        sensorManager.getDefaultSensor(
            Sensor.TYPE_ACCELEROMETER
        )

    private var canTrigger = true

    private var filteredX = 0f
    private var filteredY = 0f
    private var filteredZ = 0f

    fun start() {
        sensorManager.registerListener(
            this,
            accelerometer,
            SensorManager.SENSOR_DELAY_NORMAL
        )
    }

    fun stop() {
        sensorManager.unregisterListener(this)
    }

    override fun onSensorChanged(event: SensorEvent) {

        val alpha = 0.8f

        filteredX =
            alpha * filteredX +
                    (1 - alpha) * event.values[0]

        filteredY =
            alpha * filteredY +
                    (1 - alpha) * event.values[1]

        filteredZ =
            alpha * filteredZ +
                    (1 - alpha) * event.values[2]

        val roll = Math.toDegrees(
            atan2(
                filteredX.toDouble(),
                filteredZ.toDouble()
            )
        ).toFloat()

        val pitch = Math.toDegrees(
            atan2(
                filteredY.toDouble(),
                filteredZ.toDouble()
            )
        ).toFloat()

        processGesture(
            roll = roll,
            pitch = pitch
        )
    }

    private fun processGesture(
        roll: Float,
        pitch: Float
    ) {

        when {

            roll > 45f && canTrigger -> {
                onNextPage()
                canTrigger = false
            }

            roll < -45f && canTrigger -> {
                onPrevPage()
                canTrigger = false
            }

            pitch > 60f && canTrigger -> {
                onOpenMenu()
                canTrigger = false
            }

            pitch < -60f && canTrigger -> {
                onExitReader()
                canTrigger = false
            }

            abs(roll) < 15f &&
                    abs(pitch) < 15f -> {

                canTrigger = true
            }
        }
    }

    override fun onAccuracyChanged(
        sensor: Sensor?,
        accuracy: Int
    ) {
    }
}



========================
ReaderActivity.kt

private lateinit var tiltController: SensorTiltController


override fun onCreate(
    savedInstanceState: Bundle?
) {
    super.onCreate(savedInstanceState)

    tiltController = SensorTiltController(
        context = this,

        onNextPage = {
            readerView.nextPage()
        },

        onPrevPage = {
            readerView.prevPage()
        },

        onOpenMenu = {
            showMenu()
        },

        onExitReader = {
            finish()
        }
    )
}


override fun onResume() {
    super.onResume()
    tiltController.start()
}


override fun onPause() {
    super.onPause()
    tiltController.stop()
}



========================
动作定义
========================

向右倾斜 > 45°
↓
下一页

向左倾斜 < -45°
↓
上一页

向上抬腕 > 60°
↓
打开菜单

向下扣腕 < -60°
↓
退出阅读

恢复到中立区
(-15° ~ +15°)
↓
允许再次触发
